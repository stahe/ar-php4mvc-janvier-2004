    4. تطبيق توضيحي

    نقترح توضيح الطريقة السابقة بمثال لحساب الضرائب.

    4.1. المشكلة

    نقترح كتابة برنامج يسمح بحساب ضريبة أحد دافعي الضرائب. نضع أنفسنا في حالة مبسطة لدافع ضرائب ليس لديه سوى راتبه الوحيد ليصرح عنه:

    • نحسب عدد حصص الموظف nbParts=nbEnfants/2 +1 إذا كان غير متزوج، nbEnfants/2+2 إذا كان متزوجًا، حيث nbEnfants هو عدد أطفاله. يزداد عدد الحصص بمقدار 0.5 إذا كان هناك ثلاثة أطفال أو أكثر.
    • يتم حساب دخله الخاضع للضريبة R=0.72*S حيث S هو راتبه السنوي
    • يتم حساب معامل الأسرة Q=R/N
    • يتم حساب ضريبته I بناءً على البيانات التالية

    4.3. بنية التطبيق MVC

    سيكون للتطبيق بنية MVC التالية:

    • سيكون وحدة التحكم main.php هي وحدة التحكم العامة التي تم عرضها سابقًا
    • يتم إرسال طلب العميل إلى وحدة التحكم في شكل استعلام بالصيغة main.php?action=xx. تحدد قيمة المعلمة action البرنامج النصي للكتلة ACTIONS المطلوب تنفيذه. يعيد البرنامج النصي action الذي تم تنفيذه إلى وحدة التحكم متغيرًا يشير إلى الحالة التي يجب وضع التطبيق الويب فيها. وبناءً على هذه الحالة، ستقوم وحدة التحكم بتنشيط أحد مولدات العروض لإرسال الاستجابة إلى العميل.
    • impots-data.php هي الفئة المسؤولة عن تزويد وحدة التحكم بالبيانات التي تحتاجها
    • impots-calcul.php هي الفئة التجارية التي تسمح بحساب الضريبة

    4.4. فئة الوصول إلى البيانات

    تم إنشاء فئة الوصول إلى البيانات لإخفاء مصدر البيانات عن تطبيق الويب. في واجهتها، نجد طريقة getData التي تقدم الجداول الثلاثة للبيانات اللازمة لحساب الضريبة. في مثالنا، يتم أخذ البيانات من قاعدة بيانات MySQL. ولجعل الفئة مستقلة عن النوع الفعلي لـ SGBD، سنستخدم مكتبة pear::DB الموضحة في الملحق. وفيما يلي كود الفئة:

    <?php

   // المكتبات
  require_once 'DB.php';

  class impots_data{  
     // فئة الوصول إلى مصدر البيانات DBIMPOTS

       // السمات
        var $sDSN;         // سلسلة الاتصال
        var $sDatabase;    // اسم قاعدة البيانات
        var $oDB;          // الاتصال بقاعدة البيانات
        var $aErreurs;     // قائمة الأخطاء
        var $oRésultats;   // نتيجة الاستعلام
        var $connecté;     // قيمة منطقية تشير إلى ما إذا كان المستخدم متصلاً بالقاعدة أم لا
        var $sQuery ;      // آخر استعلام تم تنفيذه

     // منشئ
    function impots_data($dDSN){

         // $dDSN: قاموس يحدد الارتباط المطلوب إنشاؤه
       // $dDSN['sgbd']: نوع SGBD الذي يجب الاتصال به
       // $dDSN['host']: اسم الجهاز المضيف الذي يستضيفه      
       // $dDSN['database']: اسم قاعدة البيانات التي يجب الاتصال بها      
       // $dDSN['user']: أحد مستخدمي قاعدة البيانات
       // $dDSN['mdp']: كلمة المرور الخاصة به

       // ينشئ في $oDB اتصالاً بالقاعدة المحددة بواسطة $dDSN تحت هوية $dDSN['user']
       // إذا نجح الاتصال  
           // يضع في $sDSN سلسلة الاتصال بالقاعدة
           // يضع في $sDataBase اسم قاعدة البيانات التي يتم الاتصال بها
         // يضع $connecté على "صحيح"
       // إذا فشل الاتصال
           // يضع رسائل الخطأ المناسبة في القائمة $aErreurs
         // يغلق الاتصال إذا لزم الأمر
         // يضع $connecté على "خطأ" 

       // مسح قائمة الأخطاء
            $this->aErreurs=array();

       // يتم إنشاء اتصال بالقاعدة $sDSN
      $this->sDSN=$dDSN["sgbd"]."://".$dDSN["user"].":".$dDSN["mdp"]."@".$dDSN["host"]."/".$dDSN["database"];
      $this->sDatabase=$dDSN["database"];
      $this->connect();

       // متصل؟
      if( ! $this->connecté) return;

       // نجح الاتصال     
      $this->connecté=TRUE;     
    }//المصمم

    // ------------------------------------------------------------------
    function connect(){
         // (إعادة) الاتصال بالقاعدة
       // مسح قائمة الأخطاء
            $this->aErreurs=array();

       // يتم إنشاء اتصال بالقاعدة $sDSN
        $this->oDB=DB::connect($this->sDSN,true);

         // خطأ؟
        if(DB::iserror($this->oDB)){
           // تسجيل الخطأ
          $this->aErreurs[]="Echec de la connexion à la base [".$this->sDatabase."] : [".$this->oDB->getMessage()."]";
         // فشل الاتصال
        $this->connecté=FALSE;
         // النهاية
        return;
      }

       // تم الاتصال
      $this->connecté=TRUE;
    }//اتصال

    // ------------------------------------------------------------------
    function disconnect(){
       // إذا كنا متصلين، نغلق الاتصال بالقاعدة $sDSN
        if($this->connecté){
              $this->oDB->disconnect();
           // تم قطع الاتصال
        $this->connecté=FALSE;
            }//إذا
    }//disconnect

    // -------------------------------------------------------------------
    function execute($sQuery){
         // $sQuery: طلب للتنفيذ

         // يتم حفظ الطلب
            $this->sQuery=$sQuery;    

       // هل نحن متصلون؟
      if(! $this->connecté){
           // يتم تسجيل الخطأ
        $this->aErreurs[]="Pas de connexion existante à la base [$this->sDatabase]";
         // النهاية
        return;
      }//if

       // تنفيذ الطلب
      $this->oRésultats=$this->oDB->query($sQuery);

         // خطأ؟
        if(DB::iserror($this->oRésultats)){
             // يتم تسجيل الخطأ
            $this->aErreurs[]="Echec de la requête [$sQuery] : [".$this->oRésultats->getMessage()."]";
           // العودة
        return;
      }//if     
    }//تنفيذ

    // ------------------------------------------------------------------
    function getData(){
       // يتم استرداد 3 سلاسل من البيانات limits، coeffr، coeffn
      $this->execute('select limites, coeffR, coeffN from impots');
       // هل توجد أخطاء؟
      if(count($this->aErreurs)!=0) return array();
       // نستعرض نتيجة عملية الاختيار
      while ($ligne = $this->oRésultats->fetchRow(DB_FETCHMODE_ASSOC)) {
        $limites[]=$ligne['limites'];
        $coeffr[]=$ligne['coeffR'];
        $coeffn[]=$ligne['coeffN'];                
      }//while
      return array($limites,$coeffr,$coeffn);      
    }//getDataImpots

  }//فئة
?>

    4.5. فئة حساب الضريبة

    تسمح هذه الفئة بحساب ضريبة المكلف. يتم تزويد منشئها بالبيانات التي تسمح بهذا الحساب. ثم يتولى هو حساب الضريبة المقابلة. رمز الفئة هو التالي:

    <?php

  class impots_calcul{  
    // فئة حساب الضريبة

    // الشركة المصنعة
    function impots_calcul(&$perso,&$data){
      // $perso: قاموس يحتوي على المفاتيح التالية
      // الأطفال: عدد الأطفال
      // الراتب: الراتب السنوي
      // متزوج(ة): قيمة منطقية تشير إلى ما إذا كان المكلف متزوجًا أم لا
      // الضريبة (الضرائب): الضريبة المستحقة الدفع المحسوبة بواسطة هذا المُنشئ
      // $data: قاموس يحتوي على المفاتيح التالية
      // الحدود: جدول حدود الشرائح
      // coeffr: جدول معاملات الدخل
      // coeffn: جدول معاملات عدد الحصص
      // تحتوي الجداول الثلاثة على نفس عدد العناصر

      // حساب عدد الحصص
      if($perso['marié'])
        $nbParts=$perso['enfants']/2+2;
        else $nbParts=$perso['enfants']/2+1;
      if ($perso['enfants']>=3) $nbParts+=0.5;

      // الدخل الخاضع للضريبة
      $revenu=0.72*$perso['salaire'];

      // الحصة العائلية
      $QF=$revenu/$nbParts;

      // البحث عن شريحة الضرائب المطابقة لـ QF
      $nbTranches=count($data['limites']);
      $i=0;
      while($i<$nbTranches-2 && $QF>$data['limites'][$i]) $i++;

      // الضريبة
      $perso['impot']=floor($data['coeffr'][$i]*$revenu-$data['coeffn'][$i]*$nbParts);
    }//المُصنِّع
  }//الفئة
?>

    4.17. الخلاصة

    تمكنا من إظهار فائدة وحدة التحكم العامة الخاصة بنا من خلال مثال. لم نضطر إلى كتابة هذه الوحدة. اكتفينا بكتابة نصوص الإجراءات ومولدات العرض وعروض التطبيق. كما أظهرنا فائدة فصل الإجراءات عن العروض. وبذلك تمكنا من تغيير شكل الاستجابات دون تعديل سطر واحد من نصوص الإجراءات. لم يتم تعديل سوى البرامج النصية المشاركة في إنشاء العروض. ولتحقيق ذلك، يجب ألا تفترض البرامج النصية للإجراءات أي شيء بشأن العرض الذي سيعرض المعلومات التي قامت بحسابها. يجب أن تكتفي بتسليم هذه المعلومات إلى وحدة التحكم التي تنقلها إلى مولد العرض الذي سيقوم بتنسيقها. هذه قاعدة مطلقة: يجب أن تكون الإجراء منفصلة تمامًا عن العروض.
